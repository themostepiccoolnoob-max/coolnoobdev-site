ROBLOX STUDIO • ANIMATION

Custom Sprint
Animations

Create a sprint system that switches between idle, walk, and sprint animations. Hold Shift or Ctrl to sprint.

➜
MOVEMENT SYSTEM

WALK 16→ SPRINT 24

SHIFT / CTRL

1. Add the script

In Roblox Studio, add a LocalScript to StarterPlayer → StarterPlayerScripts. Replace the animation IDs below with your own uploaded animation IDs. The animations must be owned by you or shared with the experience.

LocalScript · SprintAnimations
-- SERVICES
local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")

-- PLAYER
local player = Players.LocalPlayer

-- CONFIG
local WALK_SPEED = 16
local SPRINT_SPEED = 24
local SPRINT_KEYS = {
    [Enum.KeyCode.LeftShift] = true,
    [Enum.KeyCode.LeftControl] = true,
}

-- Replace these with your animation IDs
local ANIMATION_IDS = {
    Idle = "rbxassetid://0",
    Walk = "rbxassetid://0",
    Sprint = "rbxassetid://0",
}

local function setupCharacter(character)
    local humanoid = character:WaitForChild("Humanoid")
    local animator = humanoid:WaitForChild("Animator")
    local tracks = {}

    for name, id in pairs(ANIMATION_IDS) do
        local animation = Instance.new("Animation")
        animation.AnimationId = id
        tracks[name] = animator:LoadAnimation(animation)
        tracks[name].Looped = true
    end

    local sprinting = false

    local function updateAnimation()
        if humanoid.Health <= 0 then return end

        local moving = humanoid.MoveDirection.Magnitude > 0.05
        local nextTrack

        if not moving then
            nextTrack = tracks.Idle
        elseif sprinting then
            nextTrack = tracks.Sprint
        else
            nextTrack = tracks.Walk
        end

        for _, track in pairs(tracks) do
            if track ~= nextTrack and track.IsPlaying then
                track:Stop(0.2)
            end
        end

        if nextTrack and not nextTrack.IsPlaying then
            nextTrack:Play(0.2)
        end
    end

    local function setSprinting(value)
        sprinting = value
        humanoid.WalkSpeed = sprinting and SPRINT_SPEED or WALK_SPEED
        updateAnimation()
    end

    local beganConnection = UserInputService.InputBegan:Connect(function(input, processed)
        if not processed and SPRINT_KEYS[input.KeyCode] then
            setSprinting(true)
        end
    end)

    local endedConnection = UserInputService.InputEnded:Connect(function(input)
        if SPRINT_KEYS[input.KeyCode] then
            setSprinting(false)
        end
    end)

    local runningConnection = humanoid.Running:Connect(updateAnimation)
    local diedConnection = humanoid.Died:Connect(function()
        beganConnection:Disconnect()
        endedConnection:Disconnect()
        runningConnection:Disconnect()
        for _, track in pairs(tracks) do
            track:Stop()
        end
    end)

    humanoid.WalkSpeed = WALK_SPEED
    updateAnimation()
end

if player.Character then
    setupCharacter(player.Character)
end
player.CharacterAdded:Connect(setupCharacter)

2. Set your animation IDs

Find each animation in Creator Dashboard and copy its asset ID. Replace the 0 in each rbxassetid://0 value. Set the animations to loop in the Animation Editor too, and make sure their priority is suitable (usually Movement for walk and sprint).

3. Try it in Play mode

Press Play, then hold Shift or Ctrl. If the default Roblox animation overrides yours, check that your animation was made for the rig type in your game (R6 or R15) and that the animation is permitted for the experience.

Note: This starter version uses keyboard controls. A mobile sprint button can be added separately.
